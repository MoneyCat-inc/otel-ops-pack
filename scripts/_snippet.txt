        'monitoring' {
            return @(
                'Dashboards and alerts reflect the updated thresholds with acceptable noise levels.',
                'Alert destinations receive the expected test notifications.'
            )
        }
        'development' {
            return @(
                'CI verification succeeds without regressions.',
                'Any code changes are linked back to the originating ECRR tasks.'
            )
        }
        default {
            return @(
                'Environment housekeeping commands complete without warnings.',
                'Documentation is refreshed with the latest state.'
            )
        }
    }
}

function Build-TaskContent {
    param(
        [string]$Title,
        [string]$TaskId,
        [string]$ReportRelPath,
        [string]$Category,
        [string]$Priority,
        [string]$Effort,
        [string]$Assignee,
        [string[]]$Highlights,
        [string[]]$ActionableItems,
        [string[]]$Commands,
        [string[]]$SuccessNotes
    )

    $timestamp = Get-Date -Format 'yyyy-MM-dd HH:mm:ss'
    $lines = @()

    $lines += "# Task: $Title"
    $lines += ""
    $lines += "**Task ID**: $TaskId"
    $lines += "**Created**: $timestamp UTC"
    $lines += "**Priority**: $Priority"
    $lines += "**Category**: $Category"
    $lines += "**Estimated Effort**: $Effort"
    $lines += "**Status**: pending"
    $lines += "**Assigned To**: $Assignee"
    $lines += ""
    $lines += "## Task Description"
    $lines += "Source report: $ReportRelPath"

    if ($Highlights.Count -gt 0) {
        $lines += ""
        foreach ($item in $Highlights) {
            $lines += "- $item"
        }
    }

    $lines += ""
    $lines += "## Acceptance Criteria"
    $lines += '- Complete all actions listed under Next Actions.'
    $lines += '- Run the verification commands without errors.'
    $lines += "- Document outcomes and evidence back in $ReportRelPath."
    $lines += ""
    $lines += "## Verification Commands"

    if ($Commands.Count -gt 0) {
        $lines += "```powershell"
        foreach ($cmd in $Commands) {
            $lines += $cmd
        }
        $lines += "```"
    }
    else {
        $lines += '- Review the source report for validation steps to execute.'
    }

    $lines += ""
    $lines += "## Success Metrics"
    foreach ($note in $SuccessNotes) {
        $lines += "- $note"
    }

    $lines += ""
    $lines += "## Next Actions"
    $counter = 1
    foreach ($item in $ActionableItems) {
        $lines += "$counter. $item"
        $counter++
    }

    $lines += ""
    $lines += "## Artifacts"
    $lines += "- Update $ReportRelPath with verification evidence."
    $lines += '- Store supporting logs under artifacts/ or jobs/ as required.'
    $lines += ""
    $lines += "---"
    $lines += ""
    $lines += "**Generated by**: scripts/ecrr-task-automation.ps1"
    $lines += "**Source Report**: $ReportRelPath"
    $lines += "**Generated On**: $timestamp UTC"

    return ($lines -join "`n")
}
