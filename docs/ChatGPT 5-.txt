[CmdletBinding()]
param(
  [int]$SyntheticFreshMinutes = 30
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'SilentlyContinue'

# --- helpers ---
function Mask-Secret($s) {
  if (-not $s) { return "" }
  if ($s.Length -le 8) { return ("*" * $s.Length) }
  return ($s.Substring(0,8) + "..." )
}
function Try { param($Expr) try { & $Expr } catch { $null } }
function CurlOk($url){ try { (curl -s $url | Out-String) } catch { $null } }

# ensure artifacts dir
$ArtifactsDir = Join-Path $PSScriptRoot '..\..\artifacts' | Resolve-Path -ErrorAction SilentlyContinue
if (-not $ArtifactsDir) { $ArtifactsDir = Join-Path (Get-Location) 'artifacts'; New-Item -ItemType Directory -Force -Path $ArtifactsDir | Out-Null }

# --- system info ---
$sys = [ordered]@{
  timestampUtc     = (Get-Date).ToUniversalTime().ToString("o")
  machine          = $env:COMPUTERNAME
  user             = $env:USERNAME
  os               = (Get-CimInstance Win32_OperatingSystem | Select-Object Caption,Version,BuildNumber)
  powershell       = $PSVersionTable
  culture          = (Get-Culture).Name
}

# --- env vars (masked) ---
$envs = [ordered]@{
  SIGNOZ_API_TOKEN   = Mask-Secret $env:SIGNOZ_API_TOKEN
  SIGNOZ_API_BEARER  = Mask-Secret $env:SIGNOZ_API_BEARER
  SIGNOZ_JWT         = Mask-Secret $env:SIGNOZ_JWT
  AGENT_LOG_LEVEL    = $env:AGENT_LOG_LEVEL
  WATCHDOG_LOG_LEVEL = $env:WATCHDOG_LOG_LEVEL
  LOG_LEVEL          = $env:LOG_LEVEL
}

# --- collector ---
$collector = [ordered]@{
  service   = Try { { Get-Service otelcol-contrib } }
  healthz   = (CurlOk "http://localhost:13134/healthz")  # preferred
  healthzAlt = (CurlOk "http://localhost:13133/healthz")
  configPath = "C:\otel\config.yaml"
  ports = @{
    otlpGrpc = 5317
    otlpHttp = 5318
  }
}

# --- docker / signoz ---
$docker = [ordered]@{
  ps      = Try { { docker ps --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}" } }
  chTables = Try { { docker exec -i signoz-clickhouse clickhouse-client -q "SHOW TABLES FROM signoz_logs" } }
  chCount  = Try { { docker exec -i signoz-clickhouse clickhouse-client -q "SELECT count() FROM signoz_logs.logs_v2" } }
  uiHealth = (CurlOk "http://localhost:8080/healthz")  # some builds expose /healthz
}

# --- synthetic dataset ---
$latestSynthetic = Get-ChildItem C:\logs\synthetic\*.log -ErrorAction SilentlyContinue |
  Sort-Object LastWriteTime -Descending | Select-Object -First 1
$synthetic = [ordered]@{
  latestFile = $latestSynthetic?.FullName
  length     = $latestSynthetic?.Length
  lastWrite  = $latestSynthetic?.LastWriteTime
  fresh      = if ($latestSynthetic) { ((Get-Date) - $latestSynthetic.LastWriteTime).TotalMinutes -le $SyntheticFreshMinutes } else { $false }
}

  result       = $null
  error        = $null

  } catch {
    $api.error = $_.Exception.Message
  }
} else {
  $api.error = "Skipped (no SIGNOZ_API_TOKEN)"
}

# --- WAL queue dir (optional) ---
$walPath = "C:\ProgramData\OTel\wal"
$wal = [ordered]@{
  path = $walPath
  exists = Test-Path $walPath
  sizeBytes = if (Test-Path $walPath) { (Get-ChildItem -Recurse -Force $walPath | Measure-Object -Sum Length).Sum } else { $null }
}

$report = [ordered]@{
  system      = $sys
  environment = $envs
  collector   = $collector
  docker      = $docker
  synthetic   = $synthetic
  apiProbe    = $api
  wal         = $wal
}

$md += "## System"

$md += "## Collector"

$md += "- Healthz (13133): " + ($(if ($collector.healthzAlt) { "OK" } else { "N/A" }))

$md += "## SigNoz / Docker"

$md += "```txt"
$md += ($docker.chTables | Out-String).Trim()

$md += "- logs_v2 count: $($docker.chCount)"
$md += ""
$md += "## Synthetic Dataset"
if ($synthetic.latestFile) {
  $ageMin = if ($synthetic.lastWrite) { [int]((Get-Date) - $synthetic.lastWrite).TotalMinutes } else { -1 }
  $md += "- Latest: $($synthetic.latestFile)  (age ~ ${ageMin}m, size $($synthetic.length))"
  $md += "- Fresh (≤ ${SyntheticFreshMinutes}m): " + ($(if ($synthetic.fresh) {"Yes"} else {"No"}))
} else { $md += "- No synthetic files found in C:\\logs\\synthetic" }
$md += ""
$md += "## API Probe"
$md += "- Token present: " + ($(if ($api.tokenPresent) {"Yes"} else {"No"})) + " (masked in JSON)"
if ($api.result) {
  $md += "```json"
  $md += ($api.result | ConvertTo-Json -Depth 6)
  $md += "```"
} else {
  $md += "- Result: " + ($(if ($api.error) { $api.error } else {"No result"}))
}
$md += ""
$md += "## WAL queue"
$md += "- Path: $($wal.path)  Exists: $($wal.exists)  Size(bytes): $($wal.sizeBytes)"
$md -join "`n" | Out-File -Encoding UTF8 $mdPath

Write-Host "[env-snapshot] Wrote:"
Write-Host " - $jsonPath"
Write-Host " - $mdPath"

##~yes

Awesome — here’s a safe, **one-click cleanup script** you can drop in at `scripts\env\clean.ps1`.
It’s readably logged, idempotent, and supports **dry-run** plus **full ClickHouse wipe** toggles.

```powershell
<#  scripts\env\clean.ps1
.SYNOPSIS
  Safely cleans the local Observability environment (Windows → OTel → SigNoz).
.DESCRIPTION
  Stops services/containers, removes synthetic data & generated artifacts, prunes Docker,
  and resets the OTel WAL directory. Supports dry-run and full ClickHouse wipe.

.PARAMETER Full
  Also removes SigNoz data volumes (ClickHouse, etc.). IRREVERSIBLE.

.PARAMETER KeepArtifacts
  Keep generated files in .\artifacts\ (defaults to remove *.json, *.txt)

.PARAMETER PruneDocker
  Run 'docker system prune -f' and 'docker volume prune -f'.

.PARAMETER WhatIf
  Prints actions without executing (PowerShell WhatIf semantics).

.EXAMPLES
  pwsh -File .\scripts\env\clean.ps1 -WhatIf
  pwsh -File .\scripts\env\clean.ps1 -PruneDocker
  pwsh -File .\scripts\env\clean.ps1 -Full -PruneDocker
#>

[CmdletBinding(SupportsShouldProcess)]
param(
  [switch]$Full,
  [switch]$KeepArtifacts,
  [switch]$PruneDocker
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Continue'

# --- Paths & helpers ---
$Root = Resolve-Path (Join-Path $PSScriptRoot '..\..')
Set-Location $Root

function Info  { param($m) Write-Host "[clean] $m" -ForegroundColor Cyan }
function Warn  { param($m) Write-Warning $m }
function Ok    { param($m) Write-Host "[ok] $m" -ForegroundColor Green }
function Act   { param($desc,$script)
  if ($PSCmdlet.ShouldProcess($desc)) { & $script }
}

Info "Starting cleanup in $Root"

# 1) Stop Windows OTel Collector
Act "Stop otelcol-contrib service" {
  try {
    $svc = Get-Service otelcol-contrib -ErrorAction SilentlyContinue
    if ($svc -and $svc.Status -ne 'Stopped') {
      Stop-Service otelcol-contrib -Force -ErrorAction SilentlyContinue
      $svc.WaitForStatus('Stopped','00:00:10')
    }
    Ok "otelcol-contrib service stopped (or not installed)."
  } catch { Warn "Could not stop otelcol-contrib: $($_.Exception.Message)" }
}

# 2) Stop SigNoz stack
$Compose = ".\deploy\signoz\docker-compose.yml"
if (Test-Path $Compose) {
  Act "docker compose down (SigNoz stack)" {
    & docker compose -f $Compose down 2>$null | Out-Null
    Ok "SigNoz stack down."
  }
} else {
  Warn "Compose file not found at $Compose (skipping stack shutdown)."
}

# 3) Remove synthetic logs
Act "Remove synthetic logs under C:\logs\synthetic" {
  if (Test-Path "C:\logs\synthetic") {
    Get-ChildItem "C:\logs\synthetic" -File -Recurse -ErrorAction SilentlyContinue |
      Remove-Item -Force -ErrorAction SilentlyContinue
    Ok "Synthetic logs cleared."
  } else { Ok "No synthetic logs directory found (skipped)." }
}

# 4) Remove generated artifacts (keep scripts/templates)
if (-not $KeepArtifacts) {
  Act "Remove generated artifacts (*.json, *.txt) in .\artifacts" {
    if (Test-Path ".\artifacts") {
      Get-ChildItem ".\artifacts" -Include *.json,*.txt -File -ErrorAction SilentlyContinue |
        Remove-Item -Force -ErrorAction SilentlyContinue
      Ok "Generated artifacts removed."
    } else { Ok "No artifacts directory (skipped)." }
  }
} else {
  Info "Keeping artifacts by request (-KeepArtifacts)."
}

# 5) OTel WAL reset
$Wal = "C:\ProgramData\OTel\wal"
Act "Reset WAL directory at $Wal" {
  if (Test-Path $Wal) {
    Remove-Item -Recurse -Force $Wal -ErrorAction SilentlyContinue
  }
  New-Item -ItemType Directory -Path $Wal -Force | Out-Null
  Ok "WAL directory reset."
}

# 6) Docker prune (optional)
if ($PruneDocker) {
  Act "docker system prune -f" { & docker system prune -f | Out-Null; Ok "Docker system pruned." }
  Act "docker volume prune -f" { & docker volume prune -f | Out-Null; Ok "Docker volumes pruned." }
} else {
  Info "Docker prune skipped (use -PruneDocker to enable)."
}

# 7) Full wipe: remove SigNoz volumes (ClickHouse data) – IRREVERSIBLE
if ($Full) {
  Act "Remove SigNoz volumes (full data wipe)" {
    # Remove volumes that look like SigNoz/ClickHouse; adjust filters if needed
    $vols = (docker volume ls -q | Where-Object { $_ -match 'signoz|clickhouse' })
    if ($vols) {
      foreach ($v in $vols) {
        try { & docker volume rm $v 2>$null | Out-Null } catch { }
      }
      Ok "Removed SigNoz/ClickHouse volumes: $($vols -join ', ')"
    } else { Ok "No SigNoz volumes found to remove." }
  }
} else {
  Info "Full data wipe skipped (use -Full to remove ClickHouse volumes)."
}

# 8) Summary
Info "Cleanup complete."
```
