```powershell
docker exec signoz-clickhouse clickhouse-client --query "SELECT count(*) FROM signoz_logs.distributed_logs_v2 WHERE toDateTime(timestamp/1000000000) >= now() - INTERVAL 10 MINUTE"

# Workspace cleanup (one-shot)
pwsh -File scripts\cleanup-artifacts.ps1 -Days 14           # dry-run
pwsh -File scripts\cleanup-artifacts.ps1 -Days 30 -Force    # apply

# Schedule weekly cleanup (idempotent)
pwsh -File scripts\schedule-cleanup-artifacts.ps1 -Days 30 -IncludeLogs
Get-ScheduledTask -TaskName 'OTel-Artifacts-Cleanup' | Get-ScheduledTaskInfo
```
